cmake_minimum_required(VERSION 3.22)
project(waffle_gazebo_worker LANGUAGES CXX)
find_package(gz-sim8 REQUIRED)
find_package(gz-transport13 REQUIRED)
find_package(gz-msgs10 REQUIRED)
find_package(nlohmann_json REQUIRED)
add_executable(waffle_gazebo_worker worker.cc)
target_compile_features(waffle_gazebo_worker PRIVATE cxx_std_17)
target_link_libraries(waffle_gazebo_worker PRIVATE gz-sim8::gz-sim8 gz-transport13::gz-transport13 gz-msgs10::gz-msgs10 nlohmann_json::nlohmann_json)
option(WAFFLE_ZEN4 "Tune worker code for EPYC Zen 4 (not portable)" OFF)
option(WAFFLE_LTO "Enable link time optimization for native worker" OFF)
if(WAFFLE_ZEN4)
  target_compile_options(waffle_gazebo_worker PRIVATE -march=znver4 -mtune=znver4)
endif()
if(WAFFLE_LTO)
  include(CheckIPOSupported)
  check_ipo_supported(RESULT supported OUTPUT error)
  if(NOT supported)
    message(FATAL_ERROR "LTO unavailable: ${error}")
  endif()
  set_property(TARGET waffle_gazebo_worker PROPERTY INTERPROCEDURAL_OPTIMIZATION TRUE)
endif()
install(TARGETS waffle_gazebo_worker RUNTIME DESTINATION bin)
